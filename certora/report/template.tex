% ---------- Inline code box (wrap-safe) ----------
% create an inline tcolorbox that wraps (uses \Verb for correct escaping)
\newtcbox{\inlinecode}{%
  enhanced,
  on line,
  colback=gray!25,
  colframe=gray!25,
  arc=2pt,
  outer arc=2pt,
  boxrule=0pt,
  top=1pt,
  bottom=1pt,
  left=3pt,
  right=3pt,
  boxsep=0pt,
  valign=center,
  sharp corners,
  breakable,
  fontupper=\scriptsize
}

% Replace Pandoc's \texttt with a wrap-capable inline box
% Use \Verb from fvextra (breaklines, breakanywhere) so content wraps and escapes are preserved
\renewcommand{\texttt}[1]{%
  \inlinecode{\ttfamily\strut\Verb[breaklines,breakanywhere,commandchars=\\\{\}]!#1!}%
}
